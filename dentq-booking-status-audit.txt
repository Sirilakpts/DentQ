import { NextRequest, NextResponse } from "next/server";
import { prisma } from "@/lib/prisma";
import {
  buildApprovalFlexMessage,
  buildRejectionFlexMessage,
  sendLineFlexNotification,
} from "@/lib/line";

export async function PATCH(
  req: NextRequest,
  { params }: { params: { id: string } }
) {
  try {
    const bookingId = params.id;
    const body = await req.json();
    const { status, rejectReason } = body;

    const allowedStatuses = [
      "PENDING",
      "CONFIRMED",
      "CHECKED_IN",
      "IN_PROGRESS",
      "COMPLETED",
      "CANCELLED",
      "REJECTED",
    ];

    if (!allowedStatuses.includes(status)) {
      return NextResponse.json(
        { success: false, error: "Invalid status value" },
        { status: 400 }
      );
    }

    // 1. Fetch current booking
    const booking = await prisma.booking.findUnique({
      where: { id: bookingId },
      include: { service: true },
    });

    if (!booking) {
      return NextResponse.json(
        { success: false, error: "Booking not found" },
        { status: 404 }
      );
    }

    // 2. Perform updates
    const updated = await prisma.booking.update({
      where: { id: bookingId },
      data: {
        status,
        ...(rejectReason ? { rejectReason } : {}),
      },
      include: { service: true },
    });

    let notificationSent = false;
    let notificationResult = null;

    // 3. Automated LINE Notification Triggers
    if (status === "CONFIRMED") {
      const flexMsg = buildApprovalFlexMessage({
        queueNumber: updated.queueNumber,
        patientName: updated.patientName,
        phoneNumber: updated.phoneNumber,
        date: updated.date,
        time: updated.time,
        serviceNameTh: updated.service.nameTh,
        isEmergency: updated.isEmergency,
      });

      notificationResult = await sendLineFlexNotification({
        bookingId: updated.id,
        type: "APPROVAL",
        recipient: updated.lineUserId || updated.phoneNumber,
        flexMessage: flexMsg,
      });

      await prisma.booking.update({
        where: { id: bookingId },
        data: { lineNotificationSent: true },
      });
      notificationSent = true;
    } else if (status === "REJECTED") {
      const flexMsg = buildRejectionFlexMessage({
        queueNumber: updated.queueNumber,
        patientName: updated.patientName,
        serviceNameTh: updated.service.nameTh,
        date: updated.date,
        time: updated.time,
        rejectReason: rejectReason || updated.rejectReason || undefined,
      });

      notificationResult = await sendLineFlexNotification({
        bookingId: updated.id,
        type: "REJECTION",
        recipient: updated.lineUserId || updated.phoneNumber,
        flexMessage: flexMsg,
      });

      await prisma.booking.update({
        where: { id: bookingId },
        data: { lineNotificationSent: true },
      });
      notificationSent = true;
    }

    return NextResponse.json({
      success: true,
      message: `อัปเดตสถานะเป็น ${status} สำเร็จ`,
      booking: updated,
      notificationSent,
      notificationResult,
    });
  } catch (err) {
    console.error("Error updating booking status:", err);
    return NextResponse.json(
      { success: false, error: "Failed to update booking status" },
      { status: 500 }
    );
  }
}
