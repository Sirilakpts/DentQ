"use client";

import React, { useState, useEffect } from "react";
import Link from "next/link";
import { 
  Check, 
  Calendar, 
  Clock, 
  ChevronRight, 
  ChevronLeft, 
  ShieldCheck, 
  AlertCircle, 
  User, 
  Phone, 
  FileText, 
  Sparkles, 
  Smile, 
  Activity, 
  Zap, 
  CheckCircle2, 
  Copy, 
  Share2, 
  ArrowRight,
  Info,
  Loader2
} from "lucide-react";
import confetti from "canvas-confetti";
import { formatThaiDate, getBookableDateRange } from "@/lib/utils";
import { DentalService, SlotAvailability } from "@/lib/types";

export default function BookQueuePage() {
  const [step, setStep] = useState<1 | 2 | 3 | 4>(1);

  // Data states
  const [services, setServices] = useState<DentalService[]>([]);
  const [loadingServices, setLoadingServices] = useState(true);

  // Form states
  const [selectedService, setSelectedService] = useState<DentalService | null>(null);
  
  // Date and slots
  const dateRange = getBookableDateRange();
  const [selectedDate, setSelectedDate] = useState<string>(dateRange.minDate);
  const [slotPeriod, setSlotPeriod] = useState<"ALL" | "MORNING" | "AFTERNOON">("ALL");
  const [slots, setSlots] = useState<SlotAvailability[]>([]);
  const [loadingSlots, setLoadingSlots] = useState(false);
  const [selectedSlot, setSelectedSlot] = useState<SlotAvailability | null>(null);

  // Patient Info & PDPA
  const [patientName, setPatientName] = useState("");
  const [phoneNumber, setPhoneNumber] = useState("");
  const [nationalIdOrHn, setNationalIdOrHn] = useState("");
  const [symptoms, setSymptoms] = useState("");
  const [pdpaAccepted, setPdpaAccepted] = useState(false);
  const [showPdpaModal, setShowPdpaModal] = useState(false);

  // Submission & Result
  const [submitting, setSubmitting] = useState(false);
  const [bookingError, setBookingError] = useState<string | null>(null);
  const [createdBooking, setCreatedBooking] = useState<any>(null);
  const [copied, setCopied] = useState(false);

  // 1. Fetch Services on load
  useEffect(() => {
    async function fetchServices() {
      try {
        setLoadingServices(true);
        const res = await fetch("/api/services");
        const data = await res.json();
        if (data.success) {
          setServices(data.services);
        }
      } catch (err) {
        console.error("Failed to fetch services:", err);
      } finally {
        setLoadingServices(false);
      }
    }
    fetchServices();
  }, []);

  // 2. Fetch Slots whenever selectedDate changes
  useEffect(() => {
    if (!selectedDate) return;

    async function fetchSlots() {
      try {
        setLoadingSlots(true);
        setSelectedSlot(null);
        const res = await fetch(`/api/slots?date=${selectedDate}`);
        const data = await res.json();
        if (data.success) {
          setSlots(data.slots);
        }
      } catch (err) {
        console.error("Failed to fetch slots:", err);
      } finally {
        setLoadingSlots(false);
      }
    }
    fetchSlots();
  }, [selectedDate]);

  // Handle Booking Submit
  async function handleBookingSubmit(e: React.FormEvent) {
    e.preventDefault();
    setBookingError(null);

    if (!selectedService || !selectedDate || !selectedSlot) {
      setBookingError("กรุณาเลือกบริการ วันที่ และเวลาให้ครบถ้วน");
      return;
    }

    if (!patientName.trim()) {
      setBookingError("กรุณากรอกชื่อ-นามสกุล");
      return;
    }

    const cleanPhone = phoneNumber.replace(/\D/g, "");
    if (!/^0[689]\d{8}$/.test(cleanPhone)) {
      setBookingError("กรุณากรอกเบอร์โทรศัพท์มือถือ 10 หลัก (เช่น 0812345678)");
      return;
    }

    if (!nationalIdOrHn.trim()) {
      setBookingError("กรุณากรอกเลขบัตรประชาชน 13 หลัก หรือเลข HN");
      return;
    }

    if (!pdpaAccepted) {
      setBookingError("กรุณายอมรับเงื่อนไขการคุ้มครองข้อมูลส่วนบุคคล (PDPA)");
      return;
    }

    try {
      setSubmitting(true);
      const res = await fetch("/api/bookings", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          serviceId: selectedService.id,
          date: selectedDate,
          time: selectedSlot.time,
          patientName,
          phoneNumber: cleanPhone,
          nationalIdOrHn,
          symptoms,
          pdpaAccepted: true,
        }),
      });

      const data = await res.json();

      if (!res.ok || !data.success) {
        throw new Error(data.error || "ไม่สามารถจองคิวได้ กรุณาลองใหม่อีกครั้ง");
      }

      setCreatedBooking(data.booking);
      setStep(4);

      // Trigger Confetti Celebration!
      try {
        confetti({
          particleCount: 100,
          spread: 70,
          origin: { y: 0.6 },
        });
      } catch (e) {
        // Confetti fallback
      }
    } catch (err: any) {
      setBookingError(err.message || "เกิดข้อผิดพลาดในการจองคิว");
    } finally {
      setSubmitting(false);
    }
  }

  // Filter slots by period
  const filteredSlots = slots.filter((slot) => {
    if (slotPeriod === "ALL") return true;
    return slot.period === slotPeriod;
  });

  return (
    <div className="max-w-4xl mx-auto px-4 py-8 sm:px-6">
      {/* Step Indicator Header */}
      <div className="mb-8">
        <h1 className="text-2xl sm:text-3xl font-extrabold text-slate-900 tracking-tight">
          จองคิวนัดหมายทันตกรรม
        </h1>
        <p className="text-sm text-slate-500 mt-1">
          ระบบจองคิวออนไลน์ล่วงหน้า 1-7 วัน ยืนยันรวดเร็ว พร้อมแจ้งเตือนผ่าน LINE
        </p>

        {/* Wizard Progress Steps */}
        <div className="mt-6 grid grid-cols-4 gap-2 sm:gap-4 text-xs font-medium">
          <div
            className={`flex items-center gap-2 p-2 sm:p-3 rounded-xl border ${
              step >= 1
                ? "bg-teal-50 border-teal-300 text-teal-800"
                : "bg-white border-slate-200 text-slate-400"
            }`}
          >
            <div
              className={`w-6 h-6 rounded-full flex items-center justify-center text-xs font-bold shrink-0 ${
                step > 1 ? "bg-teal-600 text-white" : "bg-teal-200 text-teal-800"
              }`}
            >
              {step > 1 ? <Check className="w-3.5 h-3.5" /> : "1"}
            </div>
            <span className="truncate hidden sm:inline">เลือกบริการ</span>
            <span className="sm:hidden">บริการ</span>
          </div>

          <div
            className={`flex items-center gap-2 p-2 sm:p-3 rounded-xl border ${
              step >= 2
                ? "bg-teal-50 border-teal-300 text-teal-800"
                : "bg-white border-slate-200 text-slate-400"
            }`}
          >
            <div
              className={`w-6 h-6 rounded-full flex items-center justify-center text-xs font-bold shrink-0 ${
                step > 2 ? "bg-teal-600 text-white" : "bg-teal-200 text-teal-800"
              }`}
            >
              {step > 2 ? <Check className="w-3.5 h-3.5" /> : "2"}
            </div>
            <span className="truncate hidden sm:inline">วัน & เวลา</span>
            <span className="sm:hidden">วันเวลา</span>
          </div>

          <div
            className={`flex items-center gap-2 p-2 sm:p-3 rounded-xl border ${
              step >= 3
                ? "bg-teal-50 border-teal-300 text-teal-800"
                : "bg-white border-slate-200 text-slate-400"
            }`}
          >
            <div
              className={`w-6 h-6 rounded-full flex items-center justify-center text-xs font-bold shrink-0 ${
                step > 3 ? "bg-teal-600 text-white" : "bg-teal-200 text-teal-800"
              }`}
            >
              {step > 3 ? <Check className="w-3.5 h-3.5" /> : "3"}
            </div>
            <span className="truncate hidden sm:inline">ข้อมูล & PDPA</span>
            <span className="sm:hidden">ข้อมูล</span>
          </div>

          <div
            className={`flex items-center gap-2 p-2 sm:p-3 rounded-xl border ${
              step === 4
                ? "bg-emerald-50 border-emerald-300 text-emerald-800"
                : "bg-white border-slate-200 text-slate-400"
            }`}
          >
            <div
              className={`w-6 h-6 rounded-full flex items-center justify-center text-xs font-bold shrink-0 ${
                step === 4 ? "bg-emerald-600 text-white" : "bg-slate-200 text-slate-500"
              }`}
            >
              4
            </div>
            <span className="truncate hidden sm:inline">บัตรคิวตรวจ</span>
            <span className="sm:hidden">บัตรคิว</span>
          </div>
        </div>
      </div>

      {/* STEP 1: SERVICE SELECTION */}
      {step === 1 && (
        <div className="bg-white rounded-2xl border border-slate-200 p-6 shadow-sm space-y-6">
          <div className="flex items-center justify-between border-b border-slate-100 pb-4">
            <div>
              <h2 className="text-lg font-bold text-slate-900">ขั้นตอนที่ 1: เลือกบริการทันตกรรม</h2>
              <p className="text-xs text-slate-500">กรุณาเลือกการรักษาที่ท่านต้องการรับบริการ</p>
            </div>
            <span className="text-xs font-semibold px-2.5 py-1 bg-teal-50 text-teal-700 rounded-full">
              ขั้นตอน 1 จาก 3
            </span>
          </div>

          {loadingServices ? (
            <div className="flex justify-center py-12">
              <Loader2 className="w-8 h-8 animate-spin text-teal-600" />
            </div>
          ) : (
            <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
              {services.map((srv) => {
                const isSelected = selectedService?.id === srv.id;
                return (
                  <div
                    key={srv.id}
                    onClick={() => setSelectedService(srv)}
                    className={`cursor-pointer rounded-xl p-5 border-2 transition-all flex flex-col justify-between ${
                      isSelected
                        ? "border-teal-500 bg-teal-50/50 shadow-sm"
                        : "border-slate-200 hover:border-teal-300 bg-white"
                    }`}
                  >
                    <div>
                      <div className="flex items-start justify-between">
                        <div className="flex items-center gap-3">
                          <div
                            className={`w-10 h-10 rounded-lg flex items-center justify-center ${
                              isSelected ? "bg-teal-600 text-white" : "bg-teal-100 text-teal-700"
                            }`}
                          >
                            {srv.icon === "Sparkles" && <Sparkles className="w-5 h-5" />}
                            {srv.icon === "Tooth" && <Activity className="w-5 h-5" />}
                            {srv.icon === "Smile" && <Smile className="w-5 h-5" />}
                            {srv.icon === "Activity" && <Zap className="w-5 h-5" />}
                          </div>
                          <div>
                            <h3 className="font-bold text-slate-900 text-sm sm:text-base">
                              {srv.nameTh}
                            </h3>
                            <p className="text-xs text-slate-400">{srv.nameEn}</p>
                          </div>
                        </div>
                        <div
                          className={`w-5 h-5 rounded-full border-2 flex items-center justify-center ${
                            isSelected ? "border-teal-600 bg-teal-600 text-white" : "border-slate-300"
                          }`}
                        >
                          {isSelected && <Check className="w-3 h-3" />}
                        </div>
                      </div>
                      <p className="text-xs text-slate-600 mt-3 leading-relaxed">
                        {srv.description}
                      </p>
                    </div>

                    <div className="mt-4 pt-3 border-t border-slate-100/80 flex items-center justify-between text-xs">
                      <span className="flex items-center gap-1 text-slate-500 font-medium">
                        <Clock className="w-3.5 h-3.5 text-teal-600" /> ประมาณ {srv.durationMinutes} นาที
                      </span>
                      <span className="font-bold text-teal-700">{srv.priceDisplay}</span>
                    </div>
                  </div>
                );
              })}
            </div>
          )}

          <div className="flex justify-end pt-4 border-t border-slate-100">
            <button
              type="button"
              disabled={!selectedService}
              onClick={() => setStep(2)}
              className="flex items-center gap-2 px-6 py-2.5 rounded-xl text-sm font-semibold text-white bg-teal-600 hover:bg-teal-700 disabled:opacity-50 disabled:cursor-not-allowed shadow-sm transition-all"
            >
              <span>ถัดไป: เลือกวันและเวลา</span>
              <ChevronRight className="w-4 h-4" />
            </button>
          </div>
        </div>
      )}

      {/* STEP 2: INTERACTIVE DATE & TIME SLOTS */}
      {step === 2 && (
        <div className="bg-white rounded-2xl border border-slate-200 p-6 shadow-sm space-y-6">
          <div className="flex items-center justify-between border-b border-slate-100 pb-4">
            <div>
              <h2 className="text-lg font-bold text-slate-900">ขั้นตอนที่ 2: เลือกวันและช่วงเวลานัดหมาย</h2>
              <p className="text-xs text-slate-500">
                บริการ: <span className="font-semibold text-teal-700">{selectedService?.nameTh}</span> ({selectedService?.durationMinutes} นาที)
              </p>
            </div>
            <span className="text-xs font-semibold px-2.5 py-1 bg-teal-50 text-teal-700 rounded-full">
              ขั้นตอน 2 จาก 3
            </span>
          </div>

          {/* Date Picker Bar (Next 7 days) */}
          <div className="space-y-2">
            <label className="block text-xs font-bold text-slate-700">
              📅 เลือกวันที่ต้องการนัดหมาย (ล่วงหน้า 1 - 7 วัน):
            </label>
            <div className="flex gap-2 overflow-x-auto pb-2 scrollbar-thin">
              {dateRange.availableDates.map((dateStr) => {
                const isSelected = selectedDate === dateStr;
                const formatted = formatThaiDate(dateStr, { short: true });
                const parts = formatted.split(" ");
                return (
                  <button
                    key={dateStr}
                    type="button"
                    onClick={() => setSelectedDate(dateStr)}
                    className={`shrink-0 flex flex-col items-center justify-center p-3 rounded-xl border min-w-[85px] transition-all ${
                      isSelected
                        ? "bg-teal-600 text-white border-teal-600 shadow-md scale-105"
                        : "bg-white text-slate-700 border-slate-200 hover:border-teal-300 hover:bg-slate-50"
                    }`}
                  >
                    <span className="text-[11px] opacity-90">{parts[1]}</span>
                    <span className="text-xl font-extrabold">{parts[0]}</span>
                    <span className="text-[10px] opacity-80">{parts[2]}</span>
                  </button>
                );
              })}
            </div>
          </div>

          {/* Time Slot Period Tabs */}
          <div className="flex items-center justify-between pt-2">
            <div className="flex gap-1.5 p-1 bg-slate-100 rounded-xl text-xs font-medium">
              <button
                type="button"
                onClick={() => setSlotPeriod("ALL")}
                className={`px-3 py-1.5 rounded-lg transition-colors ${
                  slotPeriod === "ALL" ? "bg-white text-slate-900 shadow-sm font-bold" : "text-slate-600"
                }`}
              >
                ทั้งหมด
              </button>
              <button
                type="button"
                onClick={() => setSlotPeriod("MORNING")}
                className={`px-3 py-1.5 rounded-lg transition-colors ${
                  slotPeriod === "MORNING" ? "bg-white text-teal-800 shadow-sm font-bold" : "text-slate-600"
                }`}
              >
                🌅 รอบเช้า (09:00 - 12:00)
              </button>
              <button
                type="button"
                onClick={() => setSlotPeriod("AFTERNOON")}
                className={`px-3 py-1.5 rounded-lg transition-colors ${
                  slotPeriod === "AFTERNOON" ? "bg-white text-teal-800 shadow-sm font-bold" : "text-slate-600"
                }`}
              >
                ☀️ รอบบ่าย (13:00 - 16:00)
              </button>
            </div>

            {/* Availability Legend */}
            <div className="hidden sm:flex items-center gap-3 text-[11px] text-slate-500">
              <span className="flex items-center gap-1">
                <span className="w-2.5 h-2.5 rounded-full bg-emerald-500" /> ว่าง (Available)
              </span>
              <span className="flex items-center gap-1">
                <span className="w-2.5 h-2.5 rounded-full bg-rose-500" /> เต็ม (Full)
              </span>
              <span className="flex items-center gap-1">
                <span className="w-2.5 h-2.5 rounded-full bg-slate-300" /> ระงับการจอง (Blocked)
              </span>
            </div>
          </div>

          {/* Slots Grid */}
          {loadingSlots ? (
            <div className="flex justify-center py-12">
              <Loader2 className="w-8 h-8 animate-spin text-teal-600" />
            </div>
          ) : (
            <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 gap-3">
              {filteredSlots.map((slot) => {
                const isSelected = selectedSlot?.time === slot.time;
                const currentCount = slot.currentBookings ?? 0;
                const maxCapacity = slot.maxCapacity ?? 2;
                // ข้อ 1: ตรวจสอบสถานะคิวเต็ม
                const isFull = currentCount >= maxCapacity || slot.status === "FULL";
                const isBlocked = slot.status === "BLOCKED";
                const isDisabled = isFull || isBlocked;

                return (
                  <button
                    key={slot.time}
                    type="button"
                    // ข้อ 2: ปิดการคลิกเมื่อเต็ม (Disabled State)
                    disabled={isDisabled}
                    onClick={() => setSelectedSlot(slot)}
                    // ข้อ 3 & 4: สไตล์ Tailwind CSS ตามสถานะ
                    className={`relative p-3 rounded-xl border-2 text-left transition-all flex flex-col justify-between ${
                      isSelected
                        ? "border-teal-600 bg-teal-50/80 shadow-md scale-[1.02]"
                        : isFull
                        ? "border-slate-200 bg-gray-50 opacity-70 cursor-not-allowed"
                        : isBlocked
                        ? "border-slate-200 bg-slate-100 opacity-60 cursor-not-allowed"
                        : "border-slate-200 hover:border-teal-400 hover:shadow-sm bg-white cursor-pointer"
                    }`}
                  >
                    <div className="flex items-center justify-between">
                      <span
                        className={`text-base font-extrabold ${
                          isSelected
                            ? "text-teal-900"
                            : isDisabled
                            ? "text-slate-400"
                            : "text-slate-900"
                        }`}
                      >
                        {slot.time} น.
                      </span>
                      {isSelected && (
                        <span className="w-4 h-4 rounded-full bg-teal-600 text-white flex items-center justify-center text-[10px]">
                          ✓
                        </span>
                      )}
                    </div>

                    <div className="mt-2 flex items-center justify-between text-[11px]">
                      {isBlocked ? (
                        <span className="text-slate-400 font-medium">
                          ✕ งดรับนัด
                        </span>
                      ) : isFull ? (
                        // ข้อ 3: เปลี่ยนคำว่า "ว่าง" เป็นคำว่า "เต็ม" และใช้สีแดง (text-red-500)
                        <span className="inline-flex items-center gap-1 text-red-500 font-bold">
                          <span className="w-1.5 h-1.5 rounded-full bg-red-500" />
                          เต็ม ({currentCount}/{maxCapacity})
                        </span>
                      ) : (
                        // ข้อ 4: สถานะว่างปกติ
                        <span className="inline-flex items-center gap-1 text-emerald-600 font-bold">
                          <span className="w-1.5 h-1.5 rounded-full bg-emerald-500 animate-pulse" />
                          ว่าง ({currentCount}/{maxCapacity})
                        </span>
                      )}
                    </div>
                  </button>
                );
              })}
            </div>
          )}

          {/* Action Navigation */}
          <div className="flex justify-between items-center pt-4 border-t border-slate-100">
            <button
              type="button"
              onClick={() => setStep(1)}
              className="flex items-center gap-1 px-4 py-2 text-xs font-semibold text-slate-600 hover:text-slate-900"
            >
              <ChevronLeft className="w-4 h-4" />
              <span>ย้อนกลับ</span>
            </button>

            <button
              type="button"
              disabled={!selectedSlot}
              onClick={() => setStep(3)}
              className="flex items-center gap-2 px-6 py-2.5 rounded-xl text-sm font-semibold text-white bg-teal-600 hover:bg-teal-700 disabled:opacity-50 disabled:cursor-not-allowed shadow-sm transition-all"
            >
              <span>ถัดไป: กรอกข้อมูลผู้ป่วย</span>
              <ChevronRight className="w-4 h-4" />
            </button>
          </div>
        </div>
      )}

      {/* STEP 3: PATIENT INFORMATION & PDPA CONSENT */}
      {step === 3 && (
        <form onSubmit={handleBookingSubmit} className="bg-white rounded-2xl border border-slate-200 p-6 shadow-sm space-y-6">
          <div className="flex items-center justify-between border-b border-slate-100 pb-4">
            <div>
              <h2 className="text-lg font-bold text-slate-900">ขั้นตอนที่ 3: ข้อมูลผู้ป่วย & ความยินยอม PDPA</h2>
              <p className="text-xs text-slate-500">กรุณากรอกข้อมูลที่ถูกต้องเพื่อใช้ในการยืนยันตัวตนและการแจ้งเตือน</p>
            </div>
            <span className="text-xs font-semibold px-2.5 py-1 bg-teal-50 text-teal-700 rounded-full">
              ขั้นตอน 3 จาก 3
            </span>
          </div>

          {/* Booking Summary Card */}
          <div className="bg-teal-50/70 border border-teal-200 rounded-xl p-4 flex flex-wrap items-center justify-between gap-3 text-xs">
            <div className="space-y-0.5">
              <span className="text-teal-800 font-bold text-sm">
                🦷 {selectedService?.nameTh}
              </span>
              <p className="text-teal-700">
                วัน{formatThaiDate(selectedDate, { includeDay: true })} เวลา {selectedSlot?.time} น.
              </p>
            </div>
            <div className="text-right">
              <span className="text-slate-500">ประมาณการค่าบริการ:</span>
              <p className="font-bold text-teal-900">{selectedService?.priceDisplay}</p>
            </div>
          </div>

          {bookingError && (
            <div className="p-3 rounded-xl bg-rose-50 border border-rose-200 text-rose-700 text-xs flex items-center gap-2">
              <AlertCircle className="w-4 h-4 shrink-0" />
              <span>{bookingError}</span>
            </div>
          )}

          {/* Form Inputs */}
          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
            {/* Full Name */}
            <div className="space-y-1.5">
              <label className="block text-xs font-bold text-slate-700">
                ชื่อ - นามสกุล <span className="text-rose-500">*</span>
              </label>
              <div className="relative">
                <User className="w-4 h-4 text-slate-400 absolute left-3 top-3" />
                <input
                  type="text"
                  required
                  placeholder="เช่น นาย สมชาย ใจดี"
                  value={patientName}
                  onChange={(e) => setPatientName(e.target.value)}
                  className="w-full pl-9 pr-3 py-2.5 rounded-xl border border-slate-200 text-sm focus:outline-none focus:border-teal-500 focus:ring-1 focus:ring-teal-500"
                />
              </div>
            </div>

            {/* Phone Number */}
            <div className="space-y-1.5">
              <label className="block text-xs font-bold text-slate-700">
                เบอร์โทรศัพท์มือถือ (10 หลัก) <span className="text-rose-500">*</span>
              </label>
              <div className="relative">
                <Phone className="w-4 h-4 text-slate-400 absolute left-3 top-3" />
                <input
                  type="tel"
                  required
                  maxLength={10}
                  placeholder="เช่น 0812345678"
                  value={phoneNumber}
                  onChange={(e) => setPhoneNumber(e.target.value.replace(/\D/g, ""))}
                  className="w-full pl-9 pr-3 py-2.5 rounded-xl border border-slate-200 text-sm focus:outline-none focus:border-teal-500 focus:ring-1 focus:ring-teal-500 font-mono"
                />
              </div>
              <p className="text-[11px] text-slate-400">ใช้สำหรับค้นหาบัตรคิวและรับการแจ้งเตือน</p>
            </div>

            {/* National ID / HN */}
            <div className="space-y-1.5">
              <label className="block text-xs font-bold text-slate-700">
                เลขประจำตัวประชาชน 13 หลัก <span className="text-rose-500">*</span>
              </label>
              <div className="relative">
                <FileText className="w-4 h-4 text-slate-400 absolute left-3 top-3" />
                <input
                  type="text"
                  required
                  maxLength={13}
                  placeholder="เลขบัตรประชาชน 13 หลัก"
                  value={nationalIdOrHn}
                  onChange={(e) => setNationalIdOrHn(e.target.value.replace(/\D/g, ""))}
                  className="w-full pl-9 pr-3 py-2.5 rounded-xl border border-slate-200 text-sm focus:outline-none focus:border-teal-500 focus:ring-1 focus:ring-teal-500 font-mono"
                />
              </div>
              <p className="text-[11px] text-slate-400">ข้อมูลจะถูกเข้ารหัสและปกปิดตาม พ.ร.บ. คุ้มครองข้อมูลส่วนบุคคล (PDPA)</p>
            </div>

            {/* Symptoms / Complaints */}
            <div className="space-y-1.5">
              <label className="block text-xs font-bold text-slate-700">
                อาการเบื้องต้น / ข้อมูลเพิ่มเติม (ถ้ามี)
              </label>
              <input
                type="text"
                placeholder="เช่น ปวดฟันกรามล่างขวา, เสียวฟันเวลากินน้ำเย็น"
                value={symptoms}
                onChange={(e) => setSymptoms(e.target.value)}
                className="w-full px-3 py-2.5 rounded-xl border border-slate-200 text-sm focus:outline-none focus:border-teal-500 focus:ring-1 focus:ring-teal-500"
              />
            </div>
          </div>

          {/* Mandatory PDPA Consent Card */}
          <div className="bg-slate-50 border border-slate-200 rounded-xl p-4 space-y-3">
            <div className="flex items-start gap-3">
              <input
                type="checkbox"
                id="pdpa-consent"
                required
                checked={pdpaAccepted}
                onChange={(e) => setPdpaAccepted(e.target.checked)}
                className="mt-1 w-4 h-4 text-teal-600 rounded border-slate-300 focus:ring-teal-500"
              />
              <label htmlFor="pdpa-consent" className="text-xs text-slate-700 leading-relaxed cursor-pointer">
                <span className="font-bold text-slate-900">
                  ข้าพเจ้ายินยอมให้ Q-Dentist Clinic จัดเก็บและใช้ข้อมูลส่วนบุคคล (PDPA Consent)
                </span>{" "}
                เพื่อวัตถุประสงค์ในการจองคิว จัดทำประวัติการรักษาทันตกรรม และรับการแจ้งเตือนสถานะผ่าน SMS หรือ LINE Messaging API{" "}
                <button
                  type="button"
                  onClick={() => setShowPdpaModal(true)}
                  className="text-teal-700 font-bold underline ml-1 hover:text-teal-800"
                >
                  [อ่านเงื่อนไข PDPA ฉบับเต็ม]
                </button>
              </label>
            </div>
          </div>

          {/* Action Navigation */}
          <div className="flex justify-between items-center pt-4 border-t border-slate-100">
            <button
              type="button"
              disabled={submitting}
              onClick={() => setStep(2)}
              className="flex items-center gap-1 px-4 py-2 text-xs font-semibold text-slate-600 hover:text-slate-900"
            >
              <ChevronLeft className="w-4 h-4" />
              <span>ย้อนกลับ</span>
            </button>

            <button
              type="submit"
              disabled={submitting || !pdpaAccepted}
              className="flex items-center gap-2 px-7 py-3 rounded-xl text-sm font-bold text-white bg-teal-600 hover:bg-teal-700 disabled:opacity-50 disabled:cursor-not-allowed shadow-md shadow-teal-600/30 transition-all hover:scale-[1.01]"
            >
              {submitting ? (
                <>
                  <Loader2 className="w-4 h-4 animate-spin" />
                  <span>กำลังยืนยันคิว...</span>
                </>
              ) : (
                <>
                  <CheckCircle2 className="w-4 h-4" />
                  <span>ยืนยันการจองคิวทันที</span>
                </>
              )}
            </button>
          </div>
        </form>
      )}

      {/* STEP 4: DIGITAL QUEUE TICKET & TRACKING */}
      {step === 4 && createdBooking && (
        <div className="space-y-6">
          <div className="text-center space-y-2">
            <div className="inline-flex items-center justify-center w-14 h-14 rounded-full bg-emerald-100 text-emerald-600 mb-1">
              <CheckCircle2 className="w-8 h-8" />
            </div>
            <h2 className="text-2xl font-extrabold text-slate-900">
              จองคิวสำเร็จเรียบร้อยแล้ว!
            </h2>
            <p className="text-xs sm:text-sm text-slate-500 max-w-md mx-auto">
              ระบบได้บันทึกคิวของท่านแล้ว เจ้าหน้าที่จะทำการตรวจสอบและยืนยันคิวผ่าน LINE หรือ SMS
            </p>
          </div>

          {/* Digital Queue Ticket Card */}
          <div className="max-w-md mx-auto bg-white rounded-3xl border-2 border-teal-500 shadow-xl overflow-hidden relative">
            {/* Ticket Header */}
            <div className="bg-gradient-to-r from-teal-700 to-cyan-700 p-6 text-white text-center relative">
              <div className="text-[11px] font-bold tracking-widest uppercase opacity-80">
                Q-DENTIST DIGITAL TICKET
              </div>
              <div className="text-lg font-bold mt-1">บัตรคิวนัดหมายทันตกรรม</div>
              <p className="text-xs text-teal-100 mt-0.5">คลินิกทันตกรรม สาขาพญาไท</p>

              {/* Status Badge */}
              <div className="mt-3 inline-flex items-center gap-1.5 px-3 py-1 rounded-full text-xs font-bold bg-amber-400 text-slate-900 shadow-sm">
                <span>⏳ รอเจ้าหน้าที่อนุมัติ (Pending)</span>
              </div>
            </div>

            {/* Ticket Body */}
            <div className="p-6 space-y-5 bg-white">
              {/* Big Queue Number */}
              <div className="text-center p-4 bg-teal-50 rounded-2xl border border-teal-100">
                <span className="text-xs text-slate-500 font-medium">หมายเลขคิวของคุณ</span>
                <div className="text-4xl sm:text-5xl font-black text-teal-700 tracking-tight my-1">
                  {createdBooking.queueNumber}
                </div>
                <span className="text-[11px] text-teal-600 font-semibold">
                  กรุณาแสดงบัตรคิวนี้แก่เจ้าหน้าที่
                </span>
              </div>

              {/* Ticket Details */}
              <div className="space-y-3 text-xs divide-y divide-slate-100">
                <div className="flex justify-between pt-2">
                  <span className="text-slate-500">บริการ:</span>
                  <span className="font-bold text-slate-900 text-right">{createdBooking.service?.nameTh}</span>
                </div>
                <div className="flex justify-between pt-2">
                  <span className="text-slate-500">วันนัดหมาย:</span>
                  <span className="font-bold text-slate-900">
                    {formatThaiDate(createdBooking.date, { includeDay: true })}
                  </span>
                </div>
                <div className="flex justify-between pt-2">
                  <span className="text-slate-500">เวลานัด:</span>
                  <span className="font-bold text-teal-700 text-sm">{createdBooking.time} น.</span>
                </div>
                <div className="flex justify-between pt-2">
                  <span className="text-slate-500">ชื่อผู้ป่วย:</span>
                  <span className="font-semibold text-slate-900">{createdBooking.patientName}</span>
                </div>
                <div className="flex justify-between pt-2">
                  <span className="text-slate-500">เบอร์โทรศัพท์:</span>
                  <span className="font-mono text-slate-900">{createdBooking.phoneNumber}</span>
                </div>
              </div>

              {/* Ticket Instructions */}
              <div className="bg-slate-50 p-3 rounded-xl text-[11px] text-slate-500 space-y-1">
                <p className="font-bold text-slate-700">💡 ข้อควรปฏิบัติ:</p>
                <p>• กรุณามาถึงก่อนเวลา 10-15 นาที</p>
                <p>• บันทึกภาพหน้าจอบัตรคิวนี้ไว้เพื่อความสะดวก</p>
              </div>
            </div>

            {/* Ticket Perforation effect */}
            <div className="relative flex items-center justify-between px-2 bg-white">
              <div className="w-4 h-8 bg-slate-50 rounded-r-full -ml-4 border-r border-slate-200" />
              <div className="flex-1 border-b-2 border-dashed border-slate-200 mx-2" />
              <div className="w-4 h-8 bg-slate-50 rounded-l-full -mr-4 border-l border-slate-200" />
            </div>

            {/* Ticket Footer Action Buttons */}
            <div className="p-4 bg-slate-50 flex gap-2">
              <button
                type="button"
                onClick={() => {
                  navigator.clipboard.writeText(
                    `คิวนัดหมายทำฟัน Q-Dentist หมายเลข: ${createdBooking.queueNumber} วันที่: ${createdBooking.date} เวลา: ${createdBooking.time} น.`
                  );
                  setCopied(true);
                  setTimeout(() => setCopied(false), 2000);
                }}
                className="flex-1 flex items-center justify-center gap-1.5 py-2.5 px-3 rounded-xl bg-white border border-slate-200 text-xs font-semibold text-slate-700 hover:bg-slate-100 transition-colors"
              >
                <Copy className="w-3.5 h-3.5" />
                <span>{copied ? "คัดลอกแล้ว!" : "คัดลอกข้อมูล"}</span>
              </button>

              <Link
                href={`/tracking?q=${createdBooking.queueNumber}`}
                className="flex-1 flex items-center justify-center gap-1.5 py-2.5 px-3 rounded-xl bg-teal-600 text-white text-xs font-bold hover:bg-teal-700 transition-colors"
              >
                <span>ดูในระบบติดตาม</span>
                <ArrowRight className="w-3.5 h-3.5" />
              </Link>
            </div>
          </div>

          {/* Quick Buttons below ticket */}
          <div className="text-center pt-2">
            <button
              type="button"
              onClick={() => {
                setStep(1);
                setSelectedService(null);
                setSelectedSlot(null);
                setPatientName("");
                setPhoneNumber("");
                setNationalIdOrHn("");
                setSymptoms("");
                setPdpaAccepted(false);
                setCreatedBooking(null);
              }}
              className="text-xs text-teal-700 hover:underline font-semibold"
            >
              + ทำการจองคิวใหม่เพิ่มเติม
            </button>
          </div>
        </div>
      )}

      {/* PDPA Full Terms Modal */}
      {showPdpaModal && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/50 backdrop-blur-sm animate-fade-in">
          <div className="bg-white rounded-2xl max-w-lg w-full max-h-[85vh] flex flex-col shadow-2xl">
            <div className="p-5 border-b border-slate-100 flex items-center justify-between">
              <div className="flex items-center gap-2">
                <ShieldCheck className="w-5 h-5 text-teal-600" />
                <h3 className="font-bold text-slate-900 text-base">
                  นโยบายคุ้มครองข้อมูลส่วนบุคคล (PDPA Privacy Notice)
                </h3>
              </div>
              <button
                type="button"
                onClick={() => setShowPdpaModal(false)}
                className="text-slate-400 hover:text-slate-600 text-lg"
              >
                ✕
              </button>
            </div>

            <div className="p-5 overflow-y-auto text-xs text-slate-600 space-y-3 leading-relaxed">
              <p className="font-semibold text-slate-800">
                พระราชบัญญัติคุ้มครองข้อมูลส่วนบุคคล พ.ศ. 2562 (PDPA) — คลินิกทันตกรรม Q-Dentist
              </p>
              <p>
                <strong>1. วัตถุประสงค์ในการเก็บรวบรวม:</strong> ข้อมูลชื่อ-นามสกุล, หมายเลขโทรศัพท์ และเลขประจำตัวประชาชน/HN จะถูกนำไปใช้เพื่อการระบุตัวตนของผู้รับบริการ การจัดการคิวตรวจทันตกรรม และการติดต่อแจ้งเตือนนัดหมายเท่านั้น
              </p>
              <p>
                <strong>2. การรักษาความปลอดภัยของข้อมูล:</strong> คลินิกใช้มาตรการปกป้องข้อมูลระดับมาตรฐานสากล มีการเข้ารหัสและปกปิดข้อมูล (Data Masking) บนหน้าจอสาธารณะเพื่อป้องกันการรั่วไหล
              </p>
              <p>
                <strong>3. ระยะเวลาการเก็บรักษา:</strong> ข้อมูลจะถูกเก็บรักษาตามระยะเวลาที่กฎหมายการประกอบวิชาชีพเวชกรรมและทันตกรรมกำหนด
              </p>
              <p>
                <strong>4. สิทธิของเจ้าของข้อมูลส่วนบุคคล:</strong> ท่านมีสิทธิตามกฎหมายในการขอเข้าถึง ขอสำเนา ขอแก้ไข หรือขอให้ลบข้อมูลส่วนบุคคลของท่านได้โดยแจ้งเจ้าหน้าที่คุ้มครองข้อมูลส่วนบุคคล (DPO) ของคลินิก
              </p>
            </div>

            <div className="p-4 border-t border-slate-100 flex justify-end">
              <button
                type="button"
                onClick={() => {
                  setPdpaAccepted(true);
                  setShowPdpaModal(false);
                }}
                className="px-5 py-2 rounded-xl text-xs font-bold text-white bg-teal-600 hover:bg-teal-700"
              >
                เข้าใจและยอมรับเงื่อนไข
              </button>
            </div>
          </div>
        </div>
      )}
    </div>
  );
}
